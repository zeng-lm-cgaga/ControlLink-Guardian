cmake_minimum_required(VERSION 3.16)
project(control_link_runtime_timing LANGUAGES CXX)
find_package(Threads REQUIRED)
# OBJECT 库并入各业务共享库，不新增必须安装的运行库或导出依赖。
add_library(control_link_runtime_timing OBJECT src/recorder.cpp)
set_target_properties(control_link_runtime_timing PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(control_link_runtime_timing PUBLIC cxx_std_17)
target_include_directories(control_link_runtime_timing PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(control_link_runtime_timing PRIVATE Threads::Threads)
target_compile_options(control_link_runtime_timing PRIVATE -Wall -Wextra -Wpedantic)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  enable_testing()
  add_executable(test_timing test/test_timing.cpp)
  target_link_libraries(test_timing PRIVATE control_link_runtime_timing Threads::Threads)
  target_link_options(test_timing PRIVATE -Wl,--wrap=fsync -Wl,--wrap=close)
  add_test(NAME timing_bounded_and_drain COMMAND test_timing)
endif()
