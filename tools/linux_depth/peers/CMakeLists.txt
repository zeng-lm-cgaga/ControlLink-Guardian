cmake_minimum_required(VERSION 3.16)
project(control_link_linux_peers LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(control_link_contract REQUIRED)
find_package(rclcpp REQUIRED)
get_filename_component(CONTROL_LINK_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../../.. ABSOLUTE)
set(CONTROL_LINK_ADAPTER_LIBRARY "" CACHE FILEPATH "Matching new adapter library")
add_executable(vcan_vehicle_simulator ${CONTROL_LINK_ROOT}/ros2_ws/src/control_link_bringup/tools/vcan_vehicle_simulator.cpp)
target_compile_features(vcan_vehicle_simulator PRIVATE cxx_std_17)
target_include_directories(vcan_vehicle_simulator PRIVATE ${CONTROL_LINK_ROOT}/ros2_ws/src/control_link_adapters/include)
target_link_libraries(vcan_vehicle_simulator ${CONTROL_LINK_ADAPTER_LIBRARY})
ament_target_dependencies(vcan_vehicle_simulator control_link_contract rclcpp)
target_compile_options(vcan_vehicle_simulator PRIVATE -Wall -Wextra -Wpedantic)
